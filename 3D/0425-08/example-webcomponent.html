API 文档 组件属性

📋 所有可用属性

属性名 类型 默认值 描述
sphere-color String / Number 0x3399ff 球体颜色，支持十六进制字符串或数字
wireframe-color String / Number 0x88ddff 线框颜色
star-color String / Number 0xffdd00 五角星颜色
background-color String / Number 0x0a0a1a 背景颜色
sphere-opacity Number 0.6 球体透明度 (0.0 - 1.0)
wave-speed Number 2.0 波浪动画速度
wave-amplitude Number 0.08 波浪振幅大小
show-stars Boolean true 是否显示五角星
show-wireframe Boolean true 是否显示线框
auto-rotate Boolean true 球体是否自动旋转
width String "100%" 组件宽度
height String "100%" 组件高度

使用示例 代码参考

📄 基础用法 (HTML)

<!DOCTYPE html>
<html>
<head>
    <script type="importmap">
    {
        "imports": {
            "three": "https://unpkg.com/three@0.160.0/build/three.module.js",
            "three/addons/": "https://unpkg.com/three@0.160.0/examples/jsm/"
        }
    }
    </script>
    <script type="module" src="./GlowingSphereElement.js"></script>
</head>
<body>
    <!-- 最简单的用法 -->
    <glowing-sphere style="width: 100vw; height: 100vh;"></glowing-sphere>
    
    <!-- 自定义颜色 -->
    <glowing-sphere 
        sphere-color="#ff4488"
        wireframe-color="#ff88aa"
        star-color="#ffdd44"
        background-color="#1a0a1a"
    ></glowing-sphere>
    
    <!-- 调整动画参数 -->
    <glowing-sphere 
        wave-speed="3.5"
        wave-amplitude="0.1"
        sphere-opacity="0.7"
    ></glowing-sphere>
</body>
</html>

🔧 JavaScript API 方法

const sphere = document.querySelector('glowing-sphere');

// 获取 Three.js 内部对象
const scene = sphere.getScene();
const camera = sphere.getCamera();
const renderer = sphere.getRenderer();
const mesh = sphere.getSphere();

// 手动控制（示例：添加自定义动画）
function customAnimation() {
    requestAnimationFrame(customAnimation);
    if (mesh) {
        mesh.rotation.z += 0.01;
    }
}

// 动态修改属性（会自动更新）
sphere.setAttribute('sphere-color', '#ff0000');
sphere.setAttribute('wave-speed', '5');

💚 Vue 3 组件用法

<template>
  <div class="app">
    <GlowingSphere 
      :sphere-color="sphereColor"
      :wireframe-color="wireframeColor"
      :wave-speed="waveSpeed"
      :show-stars="showStars"
      @ready="onReady"
      @error="onError"
      style="width: 100%; height: 500px;"
    />
    
    <button @click="toggleStars">
      {{ showStars ? '隐藏星星' : '显示星星' }}
    </button>
  </div>
</template>

<script setup>
import { ref } from 'vue';
import GlowingSphere from './GlowingSphere.vue';

const sphereColor = ref(0x3399ff);
const wireframeColor = ref(0x88ddff);
const waveSpeed = ref(2.0);
const showStars = ref(true);

function onReady({ scene, camera, renderer, sphere }) {
  console.log('GlowingSphere 已就绪');
  // 可以在这里添加自定义逻辑
}

function onError(error) {
  console.error('初始化失败:', error);
}

function toggleStars() {
  showStars.value = !showStars.value;
}
</script>
